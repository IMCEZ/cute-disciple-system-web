### 第8步：状态管理函数

**改什么**：实现 `updateState`、`commitState`、`getState` 三个核心函数

**怎么改**：
```javascript
// 点号路径赋值（自动创建中间对象）
function setByPath(obj, path, val) {
  const parts = path.split('.');
  let o = obj;
  for (let i = 0; i < parts.length - 1; i++) {
    if (!o[parts[i]] || typeof o[parts[i]] !== 'object') o[parts[i]] = {};
    o = o[parts[i]];
  }
  o[parts[parts.length - 1]] = val;
}

// 更新状态 + 触发渲染
function updateState(path, val) {
  setByPath(state.head, path, val);
  saveState();
  render();
}

// 增量提交（好感/余额等）
function commitState(delta) {
  // delta = { 'disciple.aff.value': +5, 'host.balance': +1000, 'host.touchLeft': -1 }
  Object.keys(delta).forEach(path => {
    const parts = path.split('.');
    const cur = getByPath(state.head, path) || 0;
    setByPath(state.head, path, cur + delta[path]);
  });
  saveState();
  render();
}
```

**效果要求**：`updateState('disciple.emotion','happy')` 能正确更新并保存

### 第9步：localStorage 持久化

**改什么**：实现 `saveState()` 和 `loadState()`

**怎么改**：
```javascript
const STORAGE_KEY = 'dz_app_state';

function saveState() {
  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } catch(e) {}
}

function loadState() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (raw) { Object.assign(state, JSON.parse(raw)); return true; }
  } catch(e) {}
  return false;
}

// 初始化时尝试加载
if (!loadState()) saveState();
```

**效果要求**：刷新页面后状态保持，修改数据后关闭重开数据不丢失

### 第10步：渲染引擎

**改什么**：实现 `render()` 函数，将 state 映射到 UI

**怎么改**：
```javascript
function render() {
  const d = state.head.disciples[0];
  const h = state.head.host;
  const m = state.head.meta;
  
  // 更新所有 data-bind 元素
  document.querySelectorAll('[data-bind]').forEach(el => {
    const path = el.getAttribute('data-bind');
    const val = getByPath(state.head, path);
    if (el.tagName === 'INPUT' || el.tagName === 'TEXTAREA') el.value = val || '';
    else el.textContent = val || '';
  });
  
  // 更新进度条
  document.querySelectorAll('[data-progress]').forEach(el => {
    const path = el.getAttribute('data-progress');
    const val = getByPath(state.head, path) || 0;
    const max = parseFloat(el.getAttribute('data-max') || '100');
    el.style.width = Math.min(100, (val / max) * 100) + '%';
  });
}
```

**效果要求**：`data-bind` 和 `data-progress` 属性的元素会自动同步数据

```javascript
let typewriterTimer = null;

function typewriter(text, containerId = 'message-flow', speed = 30) {
  return new Promise(resolve => {
    const container = document.getElementById(containerId);
    const msgDiv = document.createElement('div');
    msgDiv.className = 'msg msg-ai';
    msgDiv.innerHTML = '<div class="msg-avatar">🎭</div><div class="msg-bubble"><span class="typewriter-text"></span><span class="typewriter-cursor">▊</span></div>';
    container.appendChild(msgDiv);
    
    const textSpan = msgDiv.querySelector('.typewriter-text');
    let idx = 0;
    
    function tick() {
      if (idx < text.length) {
        textSpan.textContent += text[idx];
        idx++;
        typewriterTimer = setTimeout(tick, speed);
      } else {
        msgDiv.querySelector('.typewriter-cursor').remove();
        resolve();
      }
    }
    tick();
  });
}
```

```javascript
const API = {
  async chat(messages, options = {}) {
    const s = state.head.ui.settings;
    const provider = s.apiProvider || 'openai';
    const url = s.apiUrl || 'https://api.openai.com/v1';
    const key = s.apiKey || '';
    const model = s.model || 'gpt-4o';
    
    const body = this.buildBody(provider, model, messages, s);
    const endpoint = this.getEndpoint(provider, url, model);
    
    const res = await fetch(endpoint, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + key },
      body: JSON.stringify(body)
    });
    
    if (!res.ok) throw new Error('API Error: ' + res.status);
    return await res.json();
  },
  
  buildBody(provider, model, messages, settings) {
    const baseParams = { temperature: parseFloat(settings.temperature) || 0.9, max_tokens: parseInt(settings.maxTokens) || 2048 };
    if (provider === 'openai') return { model, messages, ...baseParams, top_p: parseFloat(settings.topP) || 0.95 };
    if (provider === 'claude') return { model, max_tokens: baseParams.max_tokens, messages, temperature: baseParams.temperature };
    if (provider === 'gemini') return { contents: messages.map(m => ({ role: m.role, parts: [{ text: m.content }] })), generationConfig: { temperature: baseParams.temperature, maxOutputTokens: baseParams.max_tokens } };
    return { model, messages, ...baseParams };
  },
  
  getEndpoint(provider, baseUrl, model) {
    if (provider === 'openai') return baseUrl.replace(/\/+$/, '') + '/chat/completions';
    if (provider === 'claude') return baseUrl.replace(/\/+$/, '') + '/messages';
    if (provider === 'gemini') return baseUrl.replace(/\/+$/, '') + '/models/' + model + ':generateContent';
    return baseUrl.replace(/\/+$/, '') + '/chat/completions';
  }
};
```

**效果要求**：封装 OpenAI/Claude/Gemini 三种协议的请求构造
